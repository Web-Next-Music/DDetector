cmake_minimum_required(VERSION 3.21)
project(UploaderTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Test)
enable_testing()
add_executable(uploader-tests ../tests/uploader_tests.cpp ../src/uploader.cpp ../src/uploader.h)
target_include_directories(uploader-tests PRIVATE ../src)
target_link_libraries(uploader-tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
add_test(NAME local-uploader COMMAND uploader-tests)
